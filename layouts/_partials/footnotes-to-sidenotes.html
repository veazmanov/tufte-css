{{- /* Turns Goldmark's footnotes into Tufte sidenotes: each [^note] reference becomes a numbered sidenote at that point in the text, and the list of footnotes at the end is dropped. A footnote of several paragraphs is joined into one, with line breaks. */ -}} {{- $c := . -}} {{- $notes := dict -}} {{- range findRESubmatch `(?s)
  • (.*?)
    • ` $c -}} {{- $body := index . 2 -}} {{- $body = replaceRE `(?s)( |\s)*]*class="footnote-backref"[^>]*>.*?` "" $body -}} {{- $body = replaceRE `(?s)

    \s*

    ` "

    " $body -}} {{- $body = replaceRE `` "" $body | strings.TrimSpace -}} {{- $notes = merge $notes (dict (index . 1) $body) -}} {{- end -}} {{- if $notes -}} {{- range findRESubmatch `[^<]*` $c -}} {{- $id := printf "sn-fn-%s" (index . 1 | anchorize) -}} {{- $note := index $notes (index . 1) | default "" -}} {{- $html := printf `%s` $id $id $note -}} {{- $c = replace $c (index . 0) $html -}} {{- end -}} {{- $c = replaceRE `(?s)

    .*
    \s*$` "" $c -}} {{- end -}} {{- return $c -}}
